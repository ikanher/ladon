```bash
TASK_REPO=/home/codex/projects/lean/matrix-factorization
TASK_THEOREM=Mf.Optimization.StochasticApproximation.Adam.ordinaryGaussian_expectedNorm_tendsto_zero
TASK_OUT=$(mktemp -d /tmp/ladon-replay-reproduction.XXXXXX)

ladon version --json > "$TASK_OUT/installed-version.json"
git -C "$TASK_REPO" rev-parse HEAD > "$TASK_OUT/source-head.txt"

  --timeout 180 --max-rss-mib 49152 --network allow --format json \
```
