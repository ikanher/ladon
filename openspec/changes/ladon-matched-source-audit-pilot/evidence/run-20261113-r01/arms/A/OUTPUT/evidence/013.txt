/-!
# Positive-part bounds under a finite measure

This owner bounds the integral of a positive part by the zeroth, first, and
second moments under the same finite measure.  Applying it to a restricted
measure preserves the restricted mass instead of introducing a zero-extension
penalty outside the retained event.
-/

open MeasureTheory

namespace Mf.DP

noncomputable section

set_option autoImplicit false

/-- Finite-measure Cauchy--Schwarz in a form that keeps the total mass. -/
theorem integral_abs_le_sqrt_measureReal_mul_integral_sq
    {X : Type*} [MeasurableSpace X]
    (measure : Measure X) [IsFiniteMeasure measure]
    (value : X → Real)
    (hvalue : AEStronglyMeasurable value measure)
    (hsquare : Integrable (fun sample => value sample ^ 2) measure) :
    (∫ sample, |value sample| ∂measure) ≤
      Real.sqrt (measure.real Set.univ * ∫ sample, value sample ^ 2 ∂measure) := by
  have hone : MemLp (fun _ : X => (1 : Real)) 2 measure :=
    (memLp_two_iff_integrable_sq
      stronglyMeasurable_const.aestronglyMeasurable).2 (integrable_const _)
  have habs : MemLp (fun sample => |value sample|) 2 measure :=
    (memLp_two_iff_integrable_sq hvalue.norm).2
      (by simpa [sq_abs] using hsquare)
  have hholder := integral_mul_le_Lp_mul_Lq_of_nonneg
    (μ := measure) (p := (2 : Real)) (q := (2 : Real))
    Real.HolderConjugate.two_two
    (ae_of_all measure fun _ : X => show 0 ≤ (1 : Real) by norm_num)
    (ae_of_all measure fun sample => abs_nonneg (value sample))
    (by simpa using hone) (by simpa using habs)
  rw [Real.sqrt_mul measureReal_nonneg]
  simpa [Real.sqrt_eq_rpow, sq_abs] using hholder

/-- The coefficient-level positive-part second-moment bound under an arbitrary
finite measure.  The square term is centered at the supplied coefficient. -/
theorem integral_positivePart_le_of_integrable_sq
    {X : Type*} [MeasurableSpace X]
    (measure : Measure X) [IsFiniteMeasure measure]
    (value : X → Real) (coefficient : Real)
    (hvalue : AEStronglyMeasurable value measure)
    (hsquare : Integrable (fun sample => (value sample - coefficient) ^ 2) measure) :
    (∫ sample, max (value sample - coefficient) 0 ∂measure) ≤
      (Real.sqrt (measure.real Set.univ *
          ∫ sample, (value sample - coefficient) ^ 2 ∂measure) +
        (∫ sample, value sample ∂measure) -
          coefficient * measure.real Set.univ) / 2 := by
  let residual : X → Real := fun sample => value sample - coefficient
  have hresidual : AEStronglyMeasurable residual measure :=
    hvalue.sub stronglyMeasurable_const.aestronglyMeasurable
  have hresidualLp : MemLp residual 2 measure :=
    (memLp_two_iff_integrable_sq hresidual).2 hsquare
  have hresidualIntegrable : Integrable residual measure :=
    hresidualLp.integrable (by norm_num)
  have hvalueIntegrable : Integrable value measure := by
    simpa [residual, sub_eq_add_neg] using
      hresidualIntegrable.add (integrable_const coefficient)
  have habs := integral_abs_le_sqrt_measureReal_mul_integral_sq
    measure residual hresidual hsquare
  have hpoint : (fun sample => max (value sample - coefficient) 0) =
      fun sample => (|residual sample| + residual sample) / 2 := by
    funext sample
    by_cases hnonneg : 0 ≤ residual sample
    · rw [max_eq_left]
      · rw [abs_of_nonneg hnonneg]
        ring
      · exact hnonneg
    · have hnonpos : residual sample ≤ 0 := le_of_not_ge hnonneg
      rw [max_eq_right hnonpos, abs_of_nonpos hnonpos]
      ring
  have habsIntegrable : Integrable (fun sample => |residual sample|) measure := by
    simpa [Real.norm_eq_abs] using hresidualIntegrable.norm
  rw [hpoint, integral_div, integral_add habsIntegrable hresidualIntegrable]
  rw [show (∫ sample, residual sample ∂measure) =
      (∫ sample, value sample ∂measure) - coefficient * measure.real Set.univ by
    dsimp [residual]
    rw [integral_sub hvalueIntegrable (integrable_const coefficient), integral_const]
    simp [smul_eq_mul, mul_comm]]
  have habs' : (∫ sample, |residual sample| ∂measure) ≤
      Real.sqrt (measure.real Set.univ *
        ∫ sample, (value sample - coefficient) ^ 2 ∂measure) := by
    simpa [residual] using habs
  linarith

/-- Expanding a square under a finite measure retains the total mass in the
constant term. -/
theorem integral_sub_const_sq_eq
    {X : Type*} [MeasurableSpace X]
    (measure : Measure X) [IsFiniteMeasure measure]
    (value : X → Real) (coefficient : Real)
    (hvalue : Integrable value measure)
    (hvalueSquare : Integrable (fun sample => value sample ^ 2) measure) :
    (∫ sample, (value sample - coefficient) ^ 2 ∂measure) =
    (∫ sample, value sample ^ 2 ∂measure) -
        2 * coefficient * (∫ sample, value sample ∂measure) +
        coefficient ^ 2 * measure.real Set.univ := by
  have hlinear : Integrable (fun sample => 2 * coefficient * value sample) measure :=
    hvalue.const_mul _
  have hconstant : Integrable (fun _ : X => coefficient ^ 2) measure :=
    integrable_const _
  rw [show (fun sample => (value sample - coefficient) ^ 2) = fun sample =>
      value sample ^ 2 - 2 * coefficient * value sample + coefficient ^ 2 by
    funext sample
    ring]
  change (∫ sample,
      ((fun x => value x ^ 2) - (fun x => 2 * coefficient * value x)) sample +
        (fun _ : X => coefficient ^ 2) sample ∂measure) = _
  rw [integral_add (hvalueSquare.sub hlinear) hconstant]
  simp only [Pi.sub_apply]
  rw [integral_sub hvalueSquare hlinear, integral_const_mul, integral_const]
  simp [smul_eq_mul, mul_comm]

/-- The centered square integral is nonnegative. -/
theorem integral_sub_const_sq_nonneg
    {X : Type*} [MeasurableSpace X]
    (measure : Measure X) (value : X → Real) (coefficient : Real) :
    0 ≤ ∫ sample, (value sample - coefficient) ^ 2 ∂measure :=
  integral_nonneg fun _ => sq_nonneg _

end

/-!
# Product sampled-Gaussian noise monotonicity

Larger coordinatewise Gaussian variance is realized by adding one independent
residual Gaussian vector to both neighboring product laws.  Common randomized
postprocessing therefore makes the exact binary profile antitone in variance
and in the public standard-deviation parameter.
-/

open MeasureTheory ProbabilityTheory

namespace Mf.DP

noncomputable section

/-- Independent centered Gaussian coordinates with a common variance. -/
def centeredGaussianProductLaw (horizon : Nat) (variance : NNReal) :
    ProbabilityMeasure (Fin horizon → Real) :=
  ProbabilityMeasure.pi (fun _ => ⟨gaussianReal 0 variance, by infer_instance⟩)

/-- Add two finite real vectors coordinatewise. -/
def coordinatewiseAdd (horizon : Nat) :
    (Fin horizon → Real) × (Fin horizon → Real) → (Fin horizon → Real) :=
  fun pair coordinate => pair.1 coordinate + pair.2 coordinate

theorem measurable_coordinatewiseAdd (horizon : Nat) :
    Measurable (coordinatewiseAdd horizon) := by
  unfold coordinatewiseAdd
  fun_prop

/-- Reassociate two finite product measures and apply the same map coordinatewise. -/
theorem map_productMeasures_coordinatewise
    {ι X Z Y : Type*} [Fintype ι]
    [MeasurableSpace X] [MeasurableSpace Z] [MeasurableSpace Y]
    (f : X × Z → Y) (hf : Measurable f)
    (mu : ι → Measure X) (rho : ι → Measure Z)
    [∀ i, IsFiniteMeasure (mu i)] [∀ i, IsFiniteMeasure (rho i)]
    [∀ i, SigmaFinite (mu i)] [∀ i, SigmaFinite (rho i)] :
    Measure.map (fun pair : (ι → X) × (ι → Z) =>
        fun i => f (pair.1 i, pair.2 i))
        ((Measure.pi mu).prod (Measure.pi rho)) =
      Measure.pi (fun i => Measure.map f ((mu i).prod (rho i))) := by
  let e : (ι → X × Z) ≃ᵐ (ι → X) × (ι → Z) :=
    MeasurableEquiv.arrowProdEquivProdArrow X Z ι
  have he : Measure.map e (Measure.pi (fun i => (mu i).prod (rho i))) =
      (Measure.pi mu).prod (Measure.pi rho) :=
    measurePreserving_arrowProdEquivProdArrow X Z ι mu rho |>.map_eq
  rw [← he, Measure.map_map (by fun_prop) e.measurable]
  have hfun :
      (fun pair : (ι → X) × (ι → Z) => fun i => f (pair.1 i, pair.2 i)) ∘ e =
        (fun sample : ι → X × Z => fun i => f (sample i)) := by
    funext sample i
    rfl
  rw [hfun, Measure.pi_map_pi (fun _ => hf.aemeasurable)]

/-- Adding a residual Gaussian vector realizes the larger-variance sampled source product. -/
theorem sampledGaussianSourceProductLaw_add_residual
    (horizon : Nat) (samplingRate : NNReal) (hRate : samplingRate ≤ 1)
    (sensitivity : Real) (small large residual : NNReal)
    (hvariance : large = small + residual) :
    Measure.map (coordinatewiseAdd horizon)
      ((sampledGaussianSourceProductLaw horizon samplingRate hRate sensitivity small).prod
        (centeredGaussianProductLaw horizon residual) :
        Measure ((Fin horizon → Real) × (Fin horizon → Real))) =
      (sampledGaussianSourceProductLaw horizon samplingRate hRate sensitivity large :
        Measure (Fin horizon → Real)) := by
  let mu : Fin horizon → Measure Real := fun _ =>
    (sampledGaussianOneStepSourceLaw samplingRate hRate sensitivity small : Measure Real)
  let rho : Fin horizon → Measure Real := fun _ => gaussianReal 0 residual
  change Measure.map (coordinatewiseAdd horizon)
      ((Measure.pi mu).prod (Measure.pi rho)) = _
  unfold coordinatewiseAdd
  rw [map_productMeasures_coordinatewise
    (fun pair : Real × Real => pair.1 + pair.2)
    (measurable_fst.add measurable_snd) mu rho]
  congr 1
  funext coordinate
  change (sampledGaussianOneStepSourceLaw samplingRate hRate sensitivity small :
      Measure Real) ∗ gaussianReal 0 residual = _
  simp [sampledGaussianOneStepSourceLaw, gaussianNoiseLaw, hvariance,
    Measure.conv_assoc, gaussianReal_conv_gaussianReal]

/-- Adding the same residual vector realizes the larger-variance reference product. -/
theorem sampledGaussianReferenceProductLaw_add_residual
    (horizon : Nat) (small large residual : NNReal)
    (hvariance : large = small + residual) :
    Measure.map (coordinatewiseAdd horizon)
      ((sampledGaussianReferenceProductLaw horizon small).prod
        (centeredGaussianProductLaw horizon residual) :
        Measure ((Fin horizon → Real) × (Fin horizon → Real))) =
      (sampledGaussianReferenceProductLaw horizon large :
        Measure (Fin horizon → Real)) := by
  simpa [sampledGaussianReferenceProductLaw, sampledGaussianSourceProductLaw,
    sampledGaussianOneStepSourceLaw_zeroSamplingRate] using
    (sampledGaussianSourceProductLaw_add_residual horizon 0 (by norm_num) 0
      small large residual hvariance)

/-- The exact sampled product profile is antitone in per-coordinate variance. -/
theorem sampledGaussianProductProfile_variance_antitone
    (horizon : Nat) (samplingRate : NNReal) (hRate : samplingRate ≤ 1)
    (sensitivity epsilon : Real) {small large : NNReal} (hvariance : small ≤ large) :
    binaryAccountingProfile
        (sampledGaussianSourceProductLaw horizon samplingRate hRate sensitivity large)
        (sampledGaussianReferenceProductLaw horizon large) epsilon ≤
      binaryAccountingProfile
        (sampledGaussianSourceProductLaw horizon samplingRate hRate sensitivity small)
        (sampledGaussianReferenceProductLaw horizon small) epsilon := by
  let residual : NNReal := large - small
  have hsum : large = small + residual := (add_tsub_cancel_of_le hvariance).symm
  have hcontract := binaryAccountingProfile_common_randomized_map_le
    (coordinatewiseAdd horizon) (measurable_coordinatewiseAdd horizon)
    (sampledGaussianSourceProductLaw horizon samplingRate hRate sensitivity small)
    (sampledGaussianReferenceProductLaw horizon small)
    (centeredGaussianProductLaw horizon residual) epsilon
  have hsource :
      ((sampledGaussianSourceProductLaw horizon samplingRate hRate sensitivity small).prod
          (centeredGaussianProductLaw horizon residual)).map
            (measurable_coordinatewiseAdd horizon).aemeasurable =
        sampledGaussianSourceProductLaw horizon samplingRate hRate sensitivity large := by
    apply Subtype.ext
    exact sampledGaussianSourceProductLaw_add_residual horizon samplingRate hRate sensitivity
      small large residual hsum
  have hreference :
      ((sampledGaussianReferenceProductLaw horizon small).prod
          (centeredGaussianProductLaw horizon residual)).map
            (measurable_coordinatewiseAdd horizon).aemeasurable =
        sampledGaussianReferenceProductLaw horizon large := by
    apply Subtype.ext
    exact sampledGaussianReferenceProductLaw_add_residual horizon small large residual hsum
  rw [hsource, hreference] at hcontract
  exact hcontract

/-- The public standard-deviation profile is antitone in noise standard deviation. -/
theorem sampledGaussianProductProfileStddev_antitone
    (horizon : Nat) (samplingRate : NNReal) (hRate : samplingRate ≤ 1)
    (sensitivity epsilon : Real) {small large : NNReal} (hStddev : small ≤ large) :
    sampledGaussianProductProfileStddev horizon samplingRate hRate sensitivity large epsilon ≤
      sampledGaussianProductProfileStddev horizon samplingRate hRate sensitivity small epsilon := by
  unfold sampledGaussianProductProfileStddev
  exact sampledGaussianProductProfile_variance_antitone horizon samplingRate hRate
    sensitivity epsilon (pow_le_pow_left' hStddev 2)

end

/-!
# Calibrated standard deviation for sampled Gaussian products

Calibration is exposed in standard-deviation units.  The underlying law
constructors continue to receive variance, so the wrapper passes the square of
the public parameter.  The infimum is total and makes no finite-horizon
attainment or positivity claim.
-/

namespace Mf.DP

noncomputable section

/-- Exact sampled-Gaussian product profile parameterized by noise standard deviation. -/
def sampledGaussianProductProfileStddev
    (horizon : Nat) (samplingRate : NNReal) (hRate : samplingRate ≤ 1)
    (sensitivity : Real) (noiseStddev : NNReal) (epsilon : Real) : ENNReal :=
  binaryAccountingProfile
    (sampledGaussianSourceProductLaw horizon samplingRate hRate sensitivity
      (noiseStddev ^ 2))
    (sampledGaussianReferenceProductLaw horizon (noiseStddev ^ 2)) epsilon

/-- Corrected public domain for sampled-Gaussian calibration. -/
def SampledGaussianCalibrationAdmissible
    (samplingRate : NNReal) (sensitivity epsilon delta : Real) : Prop :=
  0 < samplingRate ∧ samplingRate ≤ 1 ∧ 0 < sensitivity ∧
    0 ≤ epsilon ∧ 0 < delta ∧ delta < 1

/-- Standard deviations whose exact product accounting profile meets the target slack. -/
def sampledGaussianFeasibleStddevSet
    (horizon : Nat) (samplingRate : NNReal) (hRate : samplingRate ≤ 1)
    (sensitivity epsilon delta : Real) : Set NNReal :=
  {noiseStddev | sampledGaussianProductProfileStddev horizon samplingRate hRate
    sensitivity noiseStddev epsilon ≤ ENNReal.ofReal delta}

@[simp] theorem mem_sampledGaussianFeasibleStddevSet_iff
    (horizon : Nat) (samplingRate : NNReal) (hRate : samplingRate ≤ 1)
    (sensitivity epsilon delta : Real) (noiseStddev : NNReal) :
    noiseStddev ∈ sampledGaussianFeasibleStddevSet horizon samplingRate hRate
        sensitivity epsilon delta ↔
      sampledGaussianProductProfileStddev horizon samplingRate hRate sensitivity
        noiseStddev epsilon ≤ ENNReal.ofReal delta :=
  Iff.rfl

/-- Nonnegative infimum of feasible sampled-Gaussian noise standard deviations. -/
def sampledGaussianCalibratedStddev
    (horizon : Nat) (samplingRate : NNReal) (hRate : samplingRate ≤ 1)
    (sensitivity epsilon delta : Real) : NNReal :=
  sInf (sampledGaussianFeasibleStddevSet horizon samplingRate hRate
    sensitivity epsilon delta)

end

end Mf.DP
