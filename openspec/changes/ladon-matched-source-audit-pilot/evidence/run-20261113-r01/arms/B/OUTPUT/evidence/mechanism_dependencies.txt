23: /-- User replace-one adjacency through a shared dataset.  Distinct absent
24: records are removed and added, so this relation excludes the diagonal. -/
25: def UserReplaceOneAdj
26:     {Record : Type*} [DecidableEq Record]
27:     (left right : UserDataset Record) : Prop :=
28:   ∃ common removed added,
29:     removed ∉ common ∧
30:     added ∉ common ∧
31:     removed ≠ added ∧
32:     left = insert removed common ∧
33:     right = insert added common
34: 
35: theorem userReplaceOneAdj_symm
36:     {Record : Type*} [DecidableEq Record]
37:     {left right : UserDataset Record}
38:     (h : UserReplaceOneAdj left right) :
39:     UserReplaceOneAdj right left := by
40:   rcases h with ⟨common, removed, added, hremoved, hadded, hne, hleft, hright⟩
41:   exact ⟨common, added, removed, hadded, hremoved, hne.symm, hright, hleft⟩
