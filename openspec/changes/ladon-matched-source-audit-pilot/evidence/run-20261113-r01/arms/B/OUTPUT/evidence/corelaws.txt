5: # Direct sparse Gaussian likelihood ratios
6: 
7: This module keeps the privacy reference, rather than the matched Gaussian
8: source, at the origin.  Dividing a sampled Gaussian step by a positive noise
9: standard deviation sends the reference to `N(0,1)` and gives the sampled
10: source the density
11: 
12: `1 - q + q * exp (mu * z - mu^2 / 2)`
13: 
14: with respect to that standard Gaussian, where `mu` is the standardized shift.
15: -/

23: /-- Likelihood ratio of `(1-q) N(0,1) + q N(mu,1)` against `N(0,1)`. -/
24: def sparseGaussianLikelihoodRatio
25:     (samplingRate standardizedShift sample : Real) : Real :=
26:   1 - samplingRate + samplingRate *
27:     Real.exp (standardizedShift * sample - standardizedShift ^ 2 / 2)
28: 
29: /-- The unsampled mixture mass is a pointwise likelihood-ratio floor. -/
30: theorem sparseGaussianLikelihoodRatio_ge_one_sub_rate
31:     (samplingRate : NNReal) (standardizedShift sample : Real) :
32:     1 - (samplingRate : Real) ≤
33:       sparseGaussianLikelihoodRatio (samplingRate : Real) standardizedShift sample := by
34:   unfold sparseGaussianLikelihoodRatio
35:   exact le_add_of_nonneg_right
36:     (mul_nonneg (NNReal.coe_nonneg samplingRate) (Real.exp_pos _).le)
37: 
38: /-- The direct, uncentered standardized Bernoulli--Gaussian mixture. -/
39: def sparseGaussianMixtureLaw
40:     (samplingRate : NNReal) (hRate : samplingRate ≤ 1)
41:     (standardizedShift : Real) : ProbabilityMeasure Real := by
42:   let mixture : Measure Real :=
43:     ((1 - samplingRate : NNReal) : ENNReal) • gaussianReal 0 1 +
44:       (samplingRate : ENNReal) • gaussianReal standardizedShift 1
45:   refine ⟨mixture, ?_⟩
46:   constructor
47:   simp [mixture, ENNReal.coe_sub]
48:   exact tsub_add_cancel_of_le (by exact_mod_cast hRate)

54:   fun_prop

60:   unfold sparseGaussianLikelihoodRatio

21:     {Y : Type*} [MeasurableSpace Y]
22:     (mu nu : Measure Y) (epsilon : Real) : ENNReal :=
23:   ⨆ s : {s : Set Y // MeasurableSet s},
24:     mu s.1 - ENNReal.ofReal (Real.exp epsilon) * nu s.1

26: /-- An ordered event inequality with an `ENNReal` coefficient and slack. -/
27: def OrderedENNRealLawPair
28:     {Y : Type*} [MeasurableSpace Y]
29:     (mu nu : Measure Y) (coefficient slack : ENNReal) : Prop :=
30:   ∀ s, MeasurableSet s → mu s ≤ coefficient * nu s + slack

36:     eventHockeyStickProfile mu nu epsilon ≤ slack ↔

38:   rw [eventHockeyStickProfile, iSup_le_iff]
39:   constructor
40:   · intro h s hs
41:     have h' := h ⟨s, hs⟩
42:     rw [tsub_le_iff_right] at h'
43:     simpa [add_comm] using h'
