usage: ladon proof-search proof slice [-h] --repo REPO --source SOURCE
                                      --module MODULE --declaration
                                      DECLARATION --application APPLICATION
                                      [--occurrence OCCURRENCE]
                                      [--max-nodes MAX_NODES]
                                      [--max-depth MAX_DEPTH]
                                      [--timeout-seconds TIMEOUT_SECONDS]
                                      [--max-rss-mib MAX_RSS_MIB] [--scoped]
                                      [--output OUTPUT]

options:
  -h, --help            show this help message and exit
  --repo REPO
  --source SOURCE
  --module MODULE
  --declaration DECLARATION
  --application APPLICATION
  --occurrence OCCURRENCE
  --max-nodes MAX_NODES
                        Traversal node allowance, 1..1048576 (default: 131072)
  --max-depth MAX_DEPTH
                        Expression traversal depth allowance (default: 128;
                        1..4096).
  --timeout-seconds TIMEOUT_SECONDS
                        Timeout per Lean process in seconds (default: 60; not
                        total command time).
  --max-rss-mib MAX_RSS_MIB
                        Process-tree memory allowance in MiB (default: 12288).
  --scoped              Open bounded nested local scopes
  --output OUTPUT
